          return ""
      if shift_char is None:

          shift_char = chr(shift + SERIALKEY_SHIFT)  # 27

  @staticmethod
  def checksum(text):
      if not text:
          return "0"
      text = text.strip().upper()
      val = ord(text[0]) * 10         ' First char ASCII * 10
      while (val * 10) < 9999:        ' Scale up to >= 1000
          val *= 10
      total = val + sum(ord(c) for c in text)  ' Sum of ASCII + scaled
      return str(total)
