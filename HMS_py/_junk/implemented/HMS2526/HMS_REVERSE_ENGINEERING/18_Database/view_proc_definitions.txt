BillDet ### CREATE VIEW [dbo].[BillDet]
AS
SELECT     TOP 100 PERCENT dbo.PayCharge.Bill_No, SUM(dbo.PayCharge.AmtDr - dbo.PayCharge.AmtCr) AS Bill_Amt, MAX(dbo.PayCharge.SettleDate) 
                      AS BillSettleDate, MAX(dbo.PayCharge.FolioNo) AS Folio_No, MIN(dbo.GuestFolio.Vdate) AS ChkInDate, MAX(dbo.RoomOcc.ChkOutDate) AS ChkOutDate,
                       MAX(dbo.RoomOcc.ChkOutTime) AS ChkOutTime, MAX(dbo.RoomOcc.ChkOutDate) - MIN(dbo.GuestFolio.Vdate) AS Nights, 
                      MAX(dbo.GuestFolio.GuestProf) AS GuestProf, MAX(dbo.RoomOcc.RoomNo) AS ChkOutRoom, dbo.PayCharge.LogSite_Code, 
                      dbo.PayCharge.FolioNoDocid
FROM         dbo.PayCharge INNER JOIN
                      dbo.RoomOcc ON dbo.PayCharge.FolioNoDocid = dbo.RoomOcc.DocId INNER JOIN
                      dbo.GuestFolio ON dbo.RoomOcc.DocId = dbo.GuestFolio.DocId AND dbo.RoomOcc.DocId = dbo.GuestFolio.DocId
WHERE     (dbo.PayCharge.Bill_No IS NOT NULL) AND (dbo.PayCharge.Bill_No <> '') AND (dbo.RoomOcc.Type = 'O')
GROUP BY dbo.PayCharge.Bill_No, dbo.PayCharge.LogSite_Code, dbo.PayCharge.FolioNoDocid
ORDER BY MAX(dbo.PayCharge.SettleDate), dbo.PayCharge.Bill_No
