-- =====================================================
-- CHECK OUT REGISTER (FULL DETAIL) - demo columns ke saath
-- Bill No | Folio No | Guest Name | Bill Date | Mobile | In/Out Date
-- Room No. | PLAN | Occ | NIS | Goods | ROOM CHARGE | RC Per Day | EXTRA BED | MEAL CHARGE
-- CGST | SGST | TTL Tax | Discount | Round Off | Adv Mode | Advance
-- Bill Amt | P. Mode | Payment | Company | Company GSTIN | Travel Company
-- Travel GSTIN | Book No | Ref. Book Id. | Bus Source | Uname
--
-- Site: SS  |  Dates: 01/Aug/2026 se 03/Sep/2026 (neeche change karo)
--
-- NOTE: Charges (ROOM CHARGE, RC Per Day, EXTRA BED, MEAL CHARGE, CGST, SGST, etc.)
--       PayCharge se alag nikalte hain (Modeset blank hota hai),
--       Payments (P. Mode, Payment) Modeset='S' wale hain.
--       RC Per Day = ROOM CHARGE / Days  (VB6 jaisa: RoomRent = Sum(RoomCharge)/Sum(Days))
--       Days ([RC Days]) = RC voucher ke COUNT(DISTINCT Vdate). Night Audit har
--       date par ek RC voucher post karta hai (Vtype='RC', PayCode='<site>RMCH')
--       isliye per-day room charge = total RC / distinct RC dates.
--       RC Days 0 ho to NIS, phir ChkIn-ChkOut dates (DATEDIFF) se count hota hai.
--       Agar aapke site ke PayCode alag hain to top wala section
--       (RevMast codes) apne hisaab se badal lo.
--
-- NULL handling: koi bhi NULL value blank ('') dikhegi - "NULL" nahi.
-- =====================================================

DECLARE @FromDate  datetime = '2026-08-01';  -- 01/Aug/2026
DECLARE @ToDate    datetime = '2026-09-03';  -- 03/Sep/2026 (aaj tak chahiye to GETDATE() likho)
DECLARE @SiteCode  varchar(10) = 'KK';       -- apna login site code

-- Charge PayCodes (apne site ke RevMast codes se match karo):
DECLARE @RoomCharge   varchar(10) = @SiteCode + 'RMCH';  -- ROOM CHARGE
DECLARE @ExtraBed     varchar(10) = @SiteCode + 'EXTB';  -- EXTRA BED
DECLARE @MealCharge   varchar(10) = @SiteCode + 'MK';    -- MEAL CHARGE (Main Kitchen / SSTOUT)
DECLARE @CGSTCode     varchar(10) = @SiteCode + 'CGSS';  -- CGST
DECLARE @SGSTCode     varchar(10) = @SiteCode + 'SGSS';  -- SGST
DECLARE @IGSTCode     varchar(10) = @SiteCode + 'IGSS';  -- IGST
DECLARE @DiscCode     varchar(10) = @SiteCode + 'DISC';  -- Discount
DECLARE @ROFFCode     varchar(10) = @SiteCode + 'ROFF';  -- Round Off

-- NULL -> blank (''), dates -> dd-mm-yyyy
SELECT
       ISNULL(CAST([Bill No.] AS varchar(20)), '')                AS [Bill No.],
       ISNULL(CAST([Folio No.] AS varchar(30)), '')               AS [Folio No.],
       ISNULL([Guest Name], '')                                   AS [Guest Name],
       ISNULL(REPLACE(CONVERT(varchar(10), [Bill Date], 103), '/', '-'), '') AS [Bill Date],
       ISNULL([Mobile], '')                                       AS [Mobile],
       ISNULL(REPLACE(CONVERT(varchar(10), [In Date], 103), '/', '-'), '')   AS [In Date],
       ISNULL(REPLACE(CONVERT(varchar(10), [Out Date], 103), '/', '-'), '')  AS [Out Date],
       ISNULL([Room No.], '')                                     AS [Room No.],
       ISNULL([PLAN], '')                                         AS [PLAN],
       ISNULL(CAST([Occ] AS varchar(10)), '')                     AS [Occ],
       ISNULL(CAST([NIS] AS varchar(10)), '')                     AS [NIS],
       ISNULL(CAST([Goods] AS varchar(20)), '')                   AS [Goods],
ISNULL(CAST([ROOM CHARGE] AS varchar(20)), '')             AS [ROOM CHARGE],
        ISNULL(CAST(
            CASE
                WHEN ISNULL([RC Days], 0) > 0
                    THEN ROUND(ISNULL([ROOM CHARGE], 0) / CAST([RC Days] AS decimal(10,2)), 2)
                WHEN ISNULL([NIS], 0) > 0
                    THEN ROUND(ISNULL([ROOM CHARGE], 0) / CAST([NIS] AS decimal(10,2)), 2)
                WHEN [Out Date] IS NOT NULL AND [In Date] IS NOT NULL
                     AND DATEDIFF(DAY, [In Date], [Out Date]) > 0
                    THEN ROUND(ISNULL([ROOM CHARGE], 0) / DATEDIFF(DAY, [In Date], [Out Date]), 2)
                ELSE ISNULL([ROOM CHARGE], 0)
            END AS varchar(20)), '')                               AS [RC Per Day],
       ISNULL(CAST([EXTRA BED] AS varchar(20)), '')               AS [EXTRA BED],
       ISNULL(CAST([MEAL CHARGE] AS varchar(20)), '')             AS [MEAL CHARGE],
       ISNULL(CAST([CGST] AS varchar(20)), '')                    AS [CGST],
       ISNULL(CAST([SGST] AS varchar(20)), '')                    AS [SGST],
       ISNULL(CAST([TTL Tax] AS varchar(20)), '')                 AS [TTL Tax],
       ISNULL(CAST([Discount] AS varchar(20)), '')                AS [Discount],
       ISNULL(CAST([Round Off] AS varchar(20)), '')               AS [Round Off],
       ISNULL([Adv Mode], '')                                     AS [Adv Mode],
       ISNULL(CAST([Advance] AS varchar(20)), '')                 AS [Advance],
       ISNULL(CAST([Bill Amt] AS varchar(20)), '')                AS [Bill Amt],
       ISNULL([P. Mode], '')                                      AS [P. Mode],
       ISNULL(CAST([Payment] AS varchar(20)), '')                 AS [Payment],
       ISNULL([Company], '')                                      AS [Company],
       ISNULL([Company GSTIN], '')                                AS [Company GSTIN],
       ISNULL([Travel Company], '')                               AS [Travel Company],
       ISNULL([Travel GSTIN], '')                                 AS [Travel GSTIN],
       ISNULL([Book No], '')                                      AS [Book No],
       ISNULL([Ref. Book Id.], '')                                AS [Ref. Book Id.],
       ISNULL([Bus Source], '')                                   AS [Bus Source],
       ISNULL([Uname], '')                                        AS [Uname]
FROM (
SELECT
       MAX(B.Bill_No)                                        AS [Bill No.],
       MAX(B.Folio_No)                                       AS [Folio No.],
       MAX(GP.Name)                                          AS [Guest Name],
       MAX(B.BillSettleDate)                                 AS [Bill Date],
       MAX(GP.MobileNo)                                      AS [Mobile],
       MAX(B.ChkInDate)                                      AS [In Date],
       MAX(B.ChkOutDate)                                     AS [Out Date],
       MAX(B.ChkOutRoom)                                     AS [Room No.],
       MAX(PM.Name)                                          AS [PLAN],
       MAX(RO.Adult + RO.Children)                           AS [Occ],
       MAX(CAST(B.Nights AS int))                            AS [NIS],
       MAX(B.Bill_Amt)                                       AS [Goods],
(SELECT SUM(ISNULL(AmtDr,0) - ISNULL(AmtCr,0)) FROM PayCharge
         WHERE FolionoDocid = B.FolioNoDocid
           AND PayCode = @RoomCharge
           AND LogSite_Code = @SiteCode
           AND (Modeset IS NULL OR Modeset = '')                 -- charge entries (not payments)
           AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')  -- proper vtype filter
           AND (ContraDocID IS NULL OR ContraDocID = '')         -- no contra entries
         ) AS [ROOM CHARGE],
        (SELECT COUNT(DISTINCT Vdate) FROM PayCharge
         WHERE FolionoDocid = B.FolioNoDocid
           AND PayCode = @RoomCharge
           AND LogSite_Code = @SiteCode
           AND (Modeset IS NULL OR Modeset = '')
           AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')
           AND (ContraDocID IS NULL OR ContraDocID = '')
         ) AS [RC Days],
       (SELECT SUM(ISNULL(AmtDr,0) - ISNULL(AmtCr,0)) FROM PayCharge
        WHERE FolionoDocid = B.FolioNoDocid
          AND PayCode = @ExtraBed
          AND LogSite_Code = @SiteCode
          AND (Modeset IS NULL OR Modeset = '')
          AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')
          AND (ContraDocID IS NULL OR ContraDocID = '')
        ) AS [EXTRA BED],
       (SELECT SUM(ISNULL(AmtDr,0) - ISNULL(AmtCr,0)) FROM PayCharge
        WHERE FolionoDocid = B.FolioNoDocid
          AND PayCode IN (@MealCharge, @SiteCode + 'TOUT')
          AND LogSite_Code = @SiteCode
          AND (Modeset IS NULL OR Modeset = '')
          AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')
          AND (ContraDocID IS NULL OR ContraDocID = '')
        ) AS [MEAL CHARGE],
       (SELECT SUM(ISNULL(AmtDr,0) - ISNULL(AmtCr,0)) FROM PayCharge
        WHERE FolionoDocid = B.FolioNoDocid
          AND PayCode = @CGSTCode
          AND LogSite_Code = @SiteCode
          AND (Modeset IS NULL OR Modeset = '')
          AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')
          AND (ContraDocID IS NULL OR ContraDocID = '')
        ) AS [CGST],
       (SELECT SUM(ISNULL(AmtDr,0) - ISNULL(AmtCr,0)) FROM PayCharge
        WHERE FolionoDocid = B.FolioNoDocid
          AND PayCode = @SGSTCode
          AND LogSite_Code = @SiteCode
          AND (Modeset IS NULL OR Modeset = '')
          AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')
          AND (ContraDocID IS NULL OR ContraDocID = '')
        ) AS [SGST],
       (SELECT SUM(ISNULL(AmtDr,0) - ISNULL(AmtCr,0)) FROM PayCharge
        WHERE FolionoDocid = B.FolioNoDocid
          AND PayCode IN (@CGSTCode, @SGSTCode, @IGSTCode)
          AND LogSite_Code = @SiteCode
          AND (Modeset IS NULL OR Modeset = '')
          AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')
          AND (ContraDocID IS NULL OR ContraDocID = '')
        ) AS [TTL Tax],
       (SELECT SUM(ISNULL(AmtDr,0) - ISNULL(AmtCr,0)) FROM PayCharge
        WHERE FolionoDocid = B.FolioNoDocid
          AND PayCode = @DiscCode
          AND LogSite_Code = @SiteCode
          AND (Modeset IS NULL OR Modeset = '')
          AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')
          AND (ContraDocID IS NULL OR ContraDocID = '')
        ) AS [Discount],
       (SELECT SUM(ISNULL(AmtDr,0) - ISNULL(AmtCr,0)) FROM PayCharge
        WHERE FolionoDocid = B.FolioNoDocid
          AND PayCode = @ROFFCode
          AND LogSite_Code = @SiteCode
          AND (Modeset IS NULL OR Modeset = '')
          AND (Vtype NOT IN ('ARRES','ADRES','AWRES') OR DbtChkIn <> 'Yes')
          AND (ContraDocID IS NULL OR ContraDocID = '')
        ) AS [Round Off],
       MAX(ADV.PayType)                                      AS [Adv Mode],
       MAX(ADV.AdvanceAmt)                                   AS [Advance],
       MAX(B.Bill_Amt)                                       AS [Bill Amt],
       PC.PayType                                            AS [P. Mode],
       SUM(PC.AmtCr)                                         AS [Payment],
       MAX(SG.Name)                                          AS [Company],
       MAX(SG.GSTIN)                                         AS [Company GSTIN],
       MAX(TG.Name)                                          AS [Travel Company],
       MAX(TG.GSTIN)                                         AS [Travel GSTIN],
       MAX(BK.BookNo)                                        AS [Book No],
       MAX(BK.RefBookNo)                                     AS [Ref. Book Id.],
       MAX(BS.Name)                                          AS [Bus Source],
       MAX(RO.ChkoutUser)                                    AS [Uname]
FROM ((BillDet B
       INNER JOIN GuestProf GP ON B.GuestProf = GP.Code)
      INNER JOIN RoomOcc RO ON RO.DocId = B.FolioNoDocid AND RO.SNo = 1)
INNER JOIN PayCharge PC ON PC.FolionoDocid = B.FolioNoDocid AND PC.Modeset = 'S'
LEFT JOIN GuestFolio GF ON GF.DocId = B.FolioNoDocid
LEFT JOIN SubGroup  SG ON SG.SubCode = GF.Company
LEFT JOIN SubGroup  TG ON TG.SubCode = GF.TravelAgent
LEFT JOIN BussSource BS ON BS.Code = GF.BussSource
LEFT JOIN Booking   BK ON BK.DocId = GF.BookingDocId
LEFT JOIN PlanMast  PM ON PM.Code = RO.PlanCode
LEFT JOIN (SELECT FolioNoDocid, MAX(PayType) AS PayType,
                  SUM(AmtCr - AmtDr) AS AdvanceAmt
           FROM PayCharge
           WHERE Vtype IN ('REC','Chk') AND Modeset <> 'S'
             AND SettleDate BETWEEN @FromDate AND @ToDate
             AND LogSite_Code = @SiteCode
             AND (ContraDocID IS NULL OR ContraDocID = '')
           GROUP BY FolioNoDocid) ADV ON ADV.FolioNoDocid = B.FolioNoDocid
WHERE B.BillSettleDate BETWEEN @FromDate AND @ToDate
  AND B.LogSite_Code = @SiteCode
  AND PC.PayCode <> @ROFFCode                               -- off-set entry exclude
GROUP BY B.FolioNoDocid, PC.PayType
) T
ORDER BY CAST([Bill No.] AS int), [Out Date]
