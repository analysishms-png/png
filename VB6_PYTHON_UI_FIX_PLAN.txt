def _save(self):
    try:
        # NEW: Numeric validation like VB6
        amt_text = self.txt_amt.text().strip()
        if not amt_text:
            QMessageBox.warning(self, "Input", "Amount zaroori hai")
            return
        
        # Only allow numeric input (like VB6 KeyPress)
        try:
            amt = float(amt_text)
        except ValueError:
            QMessageBox.warning(self, "Input", "Sirf numeric values allowed")
            self.txt_amt.setFocus()
            return
        
        # Check against pending amount (equivalent of global_60)
        # In Python, we need to track this from the table data
        # equivalent of: CDbl(Val(CStr(Me.FgridAdjust.TextMatrix)))
        pending_amt = 0.0  # TODO: Get from equivalent of FgridAdjust row data

            # Show VB6-equivalent message
            msg = (f"Amount is Greater Then Pending Adj.Amt. "
                   f"You Can Adjust Only {pending_amt:.2f}")
            QMessageBox.warning(self, "Adjustment", msg)
            return
        
        # Proceed with save
        falo.ledgeradj_insert({
            "docid1": self.txt_docid1.text().strip(),
            "v_sno1": int(self.txt_sno1.text() or 0),
            "docid2": self.txt_docid2.text().strip(),
            "v_sno2": int(self.txt_sno2.text() or 0),
            "cr": amt,
            "subcode": self.txt_subcode.text().strip(),
            "agrefno": self.txt_agref.text().strip(),
        })
        QMessageBox.information(self, "Saved", "Adjustment saved!")
        self._load_data()
    except Exception as e:
        QMessageBox.critical(self, "Error", str(e))

def _on_subcode_change(self):

    # Search logic equivalent of VB6 Proc_183_43_EF7D54
    # Find matching ACGROUP entry and set appropriate data
    for acct in accounts:

            self.txt_subcode.setText(acct["name"])
            self.txt_subcode.setToolTip(f"SubCode: {acct['code']}")

def _load_accounts(self):
    try:
        from HMS_py.core import db
        # FIX: Use ACGROUP table like VB6, not SubGroup
        rows = db.query(
            "SELECT GroupCode, GroupName FROM ACGROUP "
            "WHERE LOGSITE_CODE='" + MemVar_1F92078 + "' OR LOGSITE_CODE='HO' "
            "ORDER BY GroupName")
        for r in rows:
            self.cmb_acct.addItem((r[1] or "").strip(), r[0])

        self.statusBar().showMessage(f"Account load error: {e}")
